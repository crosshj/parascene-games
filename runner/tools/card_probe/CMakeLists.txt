cmake_minimum_required(VERSION 3.21)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    if(DEFINED ENV{PSN00BSDK})
        set(CMAKE_TOOLCHAIN_FILE "$ENV{PSN00BSDK}/lib/libpsn00b/cmake/sdk.cmake")
    else()
        set(CMAKE_TOOLCHAIN_FILE "$ENV{HOME}/psn00bsdk/lib/libpsn00b/cmake/sdk.cmake")
    endif()
endif()
project(runner_card_probe LANGUAGES C)
psn00bsdk_add_executable(runner NOGPREL card_probe.c)
target_compile_definitions(runner PRIVATE RUNNER_WEB)
target_include_directories(runner PRIVATE ../../src)
get_filename_component(RUNNER_DIR "../.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/iso.xml" "<iso_project><track type=\"data\"><directory_tree><file name=\"SYSTEM.CNF\" type=\"data\" source=\"${RUNNER_DIR}/system.cnf\"/><file name=\"runner.exe\" type=\"data\" source=\"runner.exe\"/></directory_tree></track></iso_project>")
psn00bsdk_add_cd_image(card_probe_iso card-probe "${CMAKE_CURRENT_BINARY_DIR}/iso.xml" DEPENDS runner)
